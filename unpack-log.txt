      ClearCollect(colChStage, ["Draft", "PMC review", "Cleared to advertise", "Announced", "Certificate issued", "Interviews", "Selection approval", "Selection approved", "Offer"]);
      ClearCollect(colChPosStatus, ["Filled", "Vacant"]);
      ClearCollect(colChFunding, ["Approved", "Pending", "Not approved"]);

      ClearCollect(colChDecision, ["Waiting", "Pending", "Approved", "Returned"]);
      ClearCollect(colChRole, ["Budget", "Area fire chief", "Deputy chief", "Assistant Chief Operations", "Assistant Chief Support Services", "PMC member", "Data administrator"]);
      ClearCollect(colChAssign, ["Active", "Ended"]);
      ClearCollect(colChEvent, ["Stage change", "Selection routed", "Approval", "Detail", "Incumbent change", "Role assigned"]);
      // Who am I
      ClearCollect(colMyRoles, ForAll(Filter(FirstN('Role Assignments', 2000), Lower('User email') = varEmail && IfError(Text('Assignment status'), "") = "Active"), {role: IfError(Text(ThisRecord.Role), ""), allAreas: IfError(ThisRecord.'All areas', false), area: IfError(ThisRecord.'Geographic area'.Name, "")}));
      ClearCollect(colMyPos, Filter(Positions, 'Incumbent email' = varEmail));
      ClearCollect(colAreaNames, ForAll('Geographic Areas', {Name: ThisRecord.Name}));
      Set(varIsDAAll, CountRows(Filter(colMyRoles, role = "Data administrator" && allAreas)) > 0);
      Set(varIsDA, CountRows(Filter(colMyRoles, role = "Data administrator")) > 0);
      Set(varIsChief, CountRows(Filter(colMyRoles, role in ["Area fire chief", "Deputy chief", "Assistant Chief Operations", "Assistant Chief Support Services"])) > 0);
      Set(varIsBudget, CountRows(Filter(colMyRoles, role = "Budget")) > 0);
      Set(varIsSup, CountRows(Filter(FirstN(Positions, 2000), 'Supervisor position'.'Position ID' in colMyPos.'Position ID')) > 0);
      Set(varCanSeeAll, varIsChief || varIsDAAll);
      Set(varCanEditInc, varIsDA || varIsChief);
      Set(varCanDetail, varIsSup || varIsDA || varIsChief);
      Set(varCanEditHiring, varIsSup || varIsDA);
      Set(varCanFund, varIsBudget || varIsDA);
      // Starting area: my role's area, else my position's area
      Set(varArea, Coalesce(LookUp('Geographic Areas', Name = First(Filter(colMyRoles, !IsBlank(area))).area), LookUp('Geographic Areas', Name = First(colMyPos).'Geographic area'.Name), First('Geographic Areas')));
      // Labor estimate defaults. Pay rates and the premium are read from Dataverse when the Labor Estimate screen opens.
      Set(varPayYear, 0);
      Set(varWgPlaceholder, false);
      Set(varPrem, 32);
      Set(varShowPay, false);
      Set(varPanel, "");
      Set(varNotice, "");
      Set(varErr, "");
      Set(varSelPos, LookUp(Positions, false));
      Set(varSelAct, LookUp('Hiring Actions', false));
      ClearCollect(colDetails, Details);
      ClearCollect(colUnitOpts, {Name: "All units"}, ForAll(Filter(FirstN(Units, 2000), 'Geographic area'.Name = varArea.Name), {Name: ThisRecord.Name}))
    StartScreen: =scrTO
